From lib/schema/workout.ts (to be created in Task 1):
```ts
import { z } from "zod";

const ExerciseCoreSchema = z.object({
  exercise_name: z.string().min(1),
  sets: z.number().int().positive(),
  reps: z.string().min(1),                      // "10" | "8-12" | "AMRAP"
  rest_seconds: z.number().int().nonnegative(),
  form_cues: z.array(z.string()).default([]),
  startTimestamp: z.number().int().nonnegative().nullable(),  // SCHM-03 forward-looking
  sourceQuote: z.string().nullable(),                          // SCHM-03 forward-looking
  equipment: z.array(z.string()).default([]),                  // SCHM-03 forward-looking
});

const StandardSetSchema = ExerciseCoreSchema.extend({
  type: z.literal("standard_set"),
});

const SupersetSchema = z.object({
  type: z.literal("superset"),
  exercises: z.array(ExerciseCoreSchema).min(2),
  rest_seconds: z.number().int().nonnegative(),
});

export const RoutineItemSchema = z.discriminatedUnion("type", [
  StandardSetSchema, SupersetSchema,
]);

export const WorkoutSchema = z.object({
  schema_version: z.literal("1"),
  workout_title: z.string().min(1),
  creator_username: z.string().min(1),
  target_muscles: z.array(z.string()).default([]),
  estimated_duration_mins: z.number().int().positive(),
  difficulty: z.enum(["beginner", "intermediate", "advanced"]),
  extraction_confidence: z.enum(["high", "medium", "low"]),
  routine: z.array(RoutineItemSchema).min(1),
});

export const ExtractRequestSchema = z.object({ url: z.string().url() });

export const ExtractEventSchema = z.discriminatedUnion("type", [
  z.object({ type: z.literal("stage"), stage: z.enum(["fetching","transcribing","analyzing","generating"]) }),
  z.object({ type: z.literal("result"), workout: WorkoutSchema }),
  z.object({ type: z.literal("error"), code: z.enum(["NETWORK","NO_WORKOUT","RATE_LIMITED","UNKNOWN"]), message: z.string() }),
]);

export type Workout = z.infer<typeof WorkoutSchema>;
export type RoutineItem = z.infer<typeof RoutineItemSchema>;
export type ExtractEvent = z.infer<typeof ExtractEventSchema>;
export type ExtractRequest = z.infer<typeof ExtractRequestSchema>;
```

## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| Browser → `/api/extract` POST body | Untrusted JSON body crosses here; must be Zod-validated before any service call (ASVS V5) |
| Browser clipboard paste → URL input | Untrusted string; client-side YouTube regex + URL-tracking-param strip before submit (ASVS V5) |
| `?w=<lz-string>` query param hydration | DESERIALIZATION SURFACE — payload is lz-string-decompressed, JSON.parsed, then `WorkoutSchema.safeParse`-validated; only schema-valid output reaches React render (ASVS V5 + V7) |
| `tests/fixtures/dumbbell-leg-day.json` → module load | Build-time-trusted asset; tampering caught by `WorkoutSchema.parse()` at module init crashing the function with a clear ZodError (ASVS V14) |
| `EXTRACT_MODE` env var → service factory | Trusted Vercel env var; misconfiguration (`real` in Phase 1) is caught by the explicit `RealExtractionService` throw in `lib/extraction/real.ts` (ASVS V14) |
| Vercel Function `/api/extract` → public internet | Unauthenticated public endpoint; Phase 1 logs requests only — rate limiting and global daily cap ship in Phase 2 per ROADMAP Phase 2 scope (RESEARCH §"Security Domain / Phase 1 security gate") |
